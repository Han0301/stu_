### 10.1.1 NMS + IoU 计算

```python
import numpy as np

def compute_iou(box1, box2):
    """计算两个框的 IoU (xyxy 格式)"""
    x1 = max(box1[0], box2[0])
    y1 = max(box1[1], box2[1])
    x2 = min(box1[2], box2[2])
    y2 = min(box1[3], box2[3])

    inter = max(0, x2 - x1) * max(0, y2 - y1)
    area1 = (box1[2] - box1[0]) * (box1[3] - box1[1])
    area2 = (box2[2] - box2[0]) * (box2[3] - box2[1])
    union = area1 + area2 - inter

    return inter / union if union > 0 else 0


def nms(boxes, scores, iou_threshold):
    """非极大值抑制"""
    indices = np.argsort(scores)[::-1]
    keep = []

    while len(indices) > 0:
        i = indices[0]
        keep.append(i)

        if len(indices) == 1:
            break

        ious = np.array([compute_iou(boxes[i], boxes[j]) for j in indices[1:]])
        indices = indices[1:][ious < iou_threshold]

    return keep
```

**面试追问**：

- NMS-free 是怎么做的？（YOLOv10 的双标签分配训练）

### 10.1.2 一维卡尔曼滤波

```python
import numpy as np

class KalmanFilter1D:
    def __init__(self, x0, P0, F, H, Q, R):
        self.x = x0          # 状态 (n,)
        self.P = P0          # 协方差 (n,n)
        self.F = F           # 状态转移 (n,n)
        self.H = H           # 观测矩阵 (m,n)
        self.Q = Q           # 过程噪声 (n,n)
        self.R = R           # 观测噪声 (m,m)

    def predict(self):
        """先验预测"""
        self.x = self.F @ self.x
        self.P = self.F @ self.P @ self.F.T + self.Q
        return self.x

    def update(self, z):
        """后验更新"""
        y = z - self.H @ self.x                    # 残差 / 创新
        S = self.H @ self.P @ self.H.T + self.R    # 残差协方差
        K = self.P @ self.H.T @ np.linalg.inv(S)   # 卡尔曼增益
        self.x = self.x + K @ y                    # 状态更新
        self.P = (np.eye(len(self.x)) - K @ self.H) @ self.P  # 协方差更新
        return self.x


# 一维匀速跟踪示例
kf = KalmanFilter1D(
    x0=np.array([0.0, 0.0]),    # [位置, 速度]
    P0=np.eye(2) * 10,
    F=np.array([[1, 1], [0, 1]]),  # Δt=1
    H=np.array([[1, 0]]),
    Q=np.eye(2) * 0.1,
    R=np.array([[1.0]])
)

# 模拟观测
measurements = [1.0, 2.1, 3.0, 3.9, 5.1]
for z in measurements:
    kf.predict()
    kf.update(np.array([z]))
    print(f"位置={kf.x[0]:.2f}, 速度={kf.x[1]:.2f}")
```

**面试追问**：

- 五大公式分别是什么？（预测2步 + 更新3步）
- Q 和 R 的关系？（Q 大 → 更相信观测，R 大 → 更相信模型预测）
- EKF 与 KF 的区别？（EKF 线性化非线性系统，用 Jacobian 矩阵代替 F 和 H）
- 卡尔曼增益 K 的范围和含义？（K∈[0,1]，K 大 → 观测占比大）

### 10.1.3 旋转表示互转

```python
import numpy as np

def quaternion_to_rotation_matrix(q):
    """四元数 [w, x, y, z] → 旋转矩阵 3x3"""
    w, x, y, z = q
    return np.array([
        [1-2*y*y-2*z*z, 2*x*y-2*w*z, 2*x*z+2*w*y],
        [2*x*y+2*w*z, 1-2*x*x-2*z*z, 2*y*z-2*w*x],
        [2*x*z-2*w*y, 2*y*z+2*w*x, 1-2*x*x-2*y*y]
    ])

def rotation_matrix_to_quaternion(R):
    """旋转矩阵 → 四元数 [w, x, y, z]"""
    tr = np.trace(R)
    if tr > 0:
        S = np.sqrt(tr + 1.0) * 2
        w = 0.25 * S
        x = (R[2,1] - R[1,2]) / S
        y = (R[0,2] - R[2,0]) / S
        z = (R[1,0] - R[0,1]) / S
    elif R[0,0] > R[1,1] and R[0,0] > R[2,2]:
        S = np.sqrt(1.0 + R[0,0] - R[1,1] - R[2,2]) * 2
        w = (R[2,1] - R[1,2]) / S
        x = 0.25 * S
        y = (R[0,1] + R[1,0]) / S
        z = (R[0,2] + R[2,0]) / S
    elif R[1,1] > R[2,2]:
        S = np.sqrt(1.0 + R[1,1] - R[0,0] - R[2,2]) * 2
        w = (R[0,2] - R[2,0]) / S
        x = (R[0,1] + R[1,0]) / S
        y = 0.25 * S
        z = (R[1,2] + R[2,1]) / S
    else:
        S = np.sqrt(1.0 + R[2,2] - R[0,0] - R[1,1]) * 2
        w = (R[1,0] - R[0,1]) / S
        x = (R[0,2] + R[2,0]) / S
        y = (R[1,2] + R[2,1]) / S
        z = 0.25 * S
    return np.array([w, x, y, z])

def rotation_matrix_to_euler(R, seq='ZYX'):
    """旋转矩阵 → 欧拉角 (弧度)"""
    if seq == 'ZYX':
        sy = np.sqrt(R[0,0]**2 + R[1,0]**2)
        singular = sy < 1e-6
        if not singular:
            x = np.arctan2(R[2,1], R[2,2])
            y = np.arctan2(-R[2,0], sy)
            z = np.arctan2(R[1,0], R[0,0])
        else:
            x = np.arctan2(-R[1,2], R[1,1])
            y = np.arctan2(-R[2,0], sy)
            z = 0
        return np.array([x, y, z])  # 弧度

def euler_to_rotation_matrix(euler, seq='ZYX'):
    """欧拉角 (弧度) → 旋转矩阵"""
    x, y, z = euler
    Rx = np.array([[1,0,0],[0,np.cos(x),-np.sin(x)],[0,np.sin(x),np.cos(x)]])
    Ry = np.array([[np.cos(y),0,np.sin(y)],[0,1,0],[-np.sin(y),0,np.cos(y)]])
    Rz = np.array([[np.cos(z),-np.sin(z),0],[np.sin(z),np.cos(z),0],[0,0,1]])
    return Rz @ Ry @ Rx  # ZYX 顺序
```

### 10.2.2 本质矩阵分解

```python
def recover_pose_from_E(E, points1, points2, K):
    """从本质矩阵恢复 R,t，用正深度选唯一解"""
    U, _, Vt = np.linalg.svd(E)
    W = np.array([[0, -1, 0], [1, 0, 0], [0, 0, 1]])

    R1 = U @ W @ Vt
    R2 = U @ W.T @ Vt
    t1 = U[:, 2]
    t2 = -U[:, 2]

    # 确保旋转矩阵行列式为 +1
    if np.linalg.det(R1) < 0:
        R1 = -R1
    if np.linalg.det(R2) < 0:
        R2 = -R2

    # 构建 4 组解，用三角化点的正深度选唯一解
    best_count = -1
    best_R, best_t = None, None
    for R in [R1, R2]:
        for t in [t1, t2]:
            count = 0
            for p1, p2 in zip(points1, points2):
                # 三角化
                P1 = K @ np.eye(3, 4)
                P2 = K @ np.hstack([R, t.reshape(-1, 1)])
                X = linear_triangulation(p1, p2, P1, P2)
                if X[2] > 0 and (R @ X[:3] + t.flatten())[2] > 0:  # 两个相机前
                    count += 1
            if count > best_count:
                best_count = count
                best_R, best_t = R, t

    return best_R, best_t
```

### 10.2.3 PnP 求解（DLT 方法）

```python
def pnp_dlt(points_3d, points_2d, K):
    """
    DLT 求解 PnP
    输入: N≥6 组 3D-2D 点对, 内参 K
    输出: R (3x3), t (3,)
    """
    N = len(points_3d)
    A = np.zeros((2*N, 12))
    for i in range(N):
        X, Y, Z = points_3d[i]
        u, v = points_2d[i]
        A[2*i]   = [X, Y, Z, 1, 0,0,0,0, -u*X, -u*Y, -u*Z, -u]
        A[2*i+1] = [0,0,0,0, X, Y, Z, 1, -v*X, -v*Y, -v*Z, -v]

    _, _, Vt = np.linalg.svd(A)
    P = Vt[-1].reshape(3, 4)  # 投影矩阵 P = K [R|t]

    # 分解 P 得到 R,t
    R = np.linalg.inv(K) @ P[:, :3]
    # 强制 R 为单位正交矩阵
    U, _, Vt = np.linalg.svd(R)
    R = U @ Vt
    t = np.linalg.inv(K) @ P[:, 3]
    if np.linalg.det(R) < 0:
        R = -R
        t = -t

    return R, t
```

### 10.2.4 三角化

```python
def linear_triangulation(p1, p2, P1, P2):
    """
    线性三角化
    p1, p2: 匹配的像素坐标 (2,)
    P1, P2: 3x4 投影矩阵
    返回: 3D 齐次坐标 (4,)
    """
    A = np.array([
        p1[0] * P1[2] - P1[0],
        p1[1] * P1[2] - P1[1],
        p2[0] * P2[2] - P2[0],
        p2[1] * P2[2] - P2[1]
    ])
    _, _, Vt = np.linalg.svd(A)
    X = Vt[-1]
    return X / X[3]  # 齐次 → 非齐次
```

### 10.2.5 手写高斯滤波

```python
def gaussian_kernel(size, sigma):
    """生成高斯核"""
    ax = np.linspace(-(size-1)//2, (size-1)//2, size)
    x, y = np.meshgrid(ax, ax)
    kernel = np.exp(-(x**2 + y**2) / (2 * sigma**2))
    return kernel / kernel.sum()

def gaussian_blur(img, kernel_size=5, sigma=1.0):
    """手写高斯滤波"""
    kernel = gaussian_kernel(kernel_size, sigma)
    H, W = img.shape[:2]
    pad = kernel_size // 2

    if len(img.shape) == 2:  # 灰度
        padded = np.pad(img, pad, mode='edge')
        output = np.zeros_like(img)
        for i in range(H):
            for j in range(W):
                output[i,j] = (padded[i:i+kernel_size, j:j+kernel_size] * kernel).sum()
    else:  # 彩色
        padded = np.pad(img, ((pad,pad),(pad,pad),(0,0)), mode='edge')
        output = np.zeros_like(img)
        for i in range(H):
            for j in range(W):
                output[i,j] = (padded[i:i+kernel_size, j:j+kernel_size] * kernel[:,:,None]).sum(axis=(0,1))

    return output
```

### 10.2.6 手写 Sobel 边缘检测

```python
def sobel_edge_detection(img):
    """Sobel 边缘检测"""
    Gx = np.array([[-1, 0, 1], [-2, 0, 2], [-1, 0, 1]])
    Gy = np.array([[-1, -2, -1], [0, 0, 0], [1, 2, 1]])

    H, W = img.shape
    pad = 1
    padded = np.pad(img, pad, mode='edge')
    grad_x = np.zeros((H, W))
    grad_y = np.zeros((H, W))

    for i in range(H):
        for j in range(W):
            region = padded[i:i+3, j:j+3]
            grad_x[i,j] = (region * Gx).sum()
            grad_y[i,j] = (region * Gy).sum()

    magnitude = np.sqrt(grad_x**2 + grad_y**2)
    direction = np.arctan2(grad_y, grad_x)
    return magnitude, direction
```

## 10.4 面试回答策略

### 10.4.1 手撕代码的通用思路

```
1. 理解题目 → 复述确认（"所以你是要我做..."）
2. 沟通边界条件（输入是否可能为空？非负？）
3. 先给暴力解 → 再说优化方向
4. 写代码前先画图/列公式
5. 写代码时边写边解释思路
6. 写完后手动推导一个简单测试 case
7. 分析时空复杂度
```

### 10.4.2 视觉推导题的答题框架

```text
以"PnP 求解"为例：
    1. 问题定义（已知 N 个 3D-2D 点对，求 [R|t]）
    2. 列出核心公式 z*[u,v,1]^T = K[R|t][X,Y,Z,1]^T
    3. 解法选择：
       - N=3: P3P（解多项式，4 组解）
       - N≥6: DLT（线性最小二乘）
       - 大多情况: EPnP + RANSAC（鲁棒）
       - 最终优化: L-M 迭代（最小化重投影误差）
    4. 画图示意几何关系
    5. 写出关键代码/推导步骤
    6. 指出可能的坑（RANSAC 迭代次数、平面退化情况）
```
