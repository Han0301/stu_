### 1.2 链表——跟踪目标的生命周期管理

```python
class TrackNode:
    """双向链表节点——管理单个跟踪目标的生命周期"""
    def __init__(self, track_id: int):
        self.track_id = track_id
        self.history: list = []       # 轨迹历史
        self.age: int = 0             # 存活帧数
        self.lost_count: int = 0      # 连续丢失帧数
        self.prev: 'TrackNode' = None  # 前驱
        self.next: 'TrackNode' = None  # 后继

class TrackletList:
    """双向链表管理所有跟踪目标"""
    def __init__(self, max_lost=30):
        self.head = None
        self.tail = None
        self.size = 0
        self.max_lost = max_lost  # 丢失超过 30 帧则删除
    
    def add_track(self, track_id: int):
        """添加新目标到链表尾部"""
        node = TrackNode(track_id)
        if self.tail is None:
            self.head = self.tail = node
        else:
            self.tail.next = node
            node.prev = self.tail
            self.tail = node
        self.size += 1
        return node
    
    def remove_track(self, node: TrackNode):
        """从链表中删除目标——O(1)"""
        if node.prev:
            node.prev.next = node.next
        if node.next:
            node.next.prev = node.prev
        if node == self.head:
            self.head = node.next
        if node == self.tail:
            self.tail = node.prev
        self.size -= 1
    
    def clean_lost(self):
        """清理超过最大丢失帧数的目标"""
        curr = self.head
        while curr:
            next_node = curr.next
            if curr.lost_count > self.max_lost:
                self.remove_track(curr)
            curr = next_node
```

### 2.2 队列（Queue）——FIFO

```python
from collections import deque

# BFS 遍历图像——连通域分析
def bfs_connected_components(binary_img):
    """用 BFS 标记连通区域"""
    h, w = binary_img.shape
    visited = np.zeros_like(binary_img, dtype=bool)
    q = deque()
    components = []
    
    for y in range(h):
        for x in range(w):
            if binary_img[y, x] and not visited[y, x]:
                # 发现新连通域
                component = []
                q.append((x, y))
                visited[y, x] = True
                
                while q:
                    cx, cy = q.popleft()  # FIFO
                    component.append((cx, cy))
                    
                    # 四邻域搜索
                    for dx, dy in [(1,0), (-1,0), (0,1), (0,-1)]:
                        nx, ny = cx+dx, cy+dy
                        if 0 <= nx < w and 0 <= ny < h:
                            if binary_img[ny, nx] and not visited[ny, nx]:
                                visited[ny, nx] = True
                                q.append((nx, ny))
                
                components.append(component)
    
    return components
```

### 2.3 优先队列（Priority Queue / Heap）——NMS

```python
import heapq

def nms_with_heap(detections, iou_threshold=0.5):
    """
    用最大堆实现 NMS
    detections: [(x1, y1, x2, y2, score), ...]
    时间复杂度 O(n log n)
    """
    # Python heapq 是最小堆，存负值模拟最大堆
    heap = [(-score, x1, y1, x2, y2) 
            for x1, y1, x2, y2, score in detections]
    heapq.heapify(heap)
    
    keep = []
    while heap:
        # 取置信度最高的框
        neg_score, *box = heapq.heappop(heap)
        keep.append(box)
        
        # 重新建堆——移出与当前框 IoU 过高的
        # （实际 NMS 有更高效的实现，这里展示堆的思想）
        remaining = []
        for item in heap:
            _, *other = item
            if compute_iou(box, other) < iou_threshold:
                remaining.append(item)
        heap = remaining
        heapq.heapify(heap)
    
    return keep
```

### 2.4 双端队列（Deque）——滑动窗口

```python
from collections import deque

# 滑动窗口均值滤波
def sliding_window_average(data, window_size=5):
    """对时序数据做滑动窗口平均"""
    window = deque(maxlen=window_size)
    result = []
    
    for value in data:
        window.append(value)
        result.append(sum(window) / len(window))
    
    return result

# 应用：平滑检测框的抖动
raw_boxes = [(100, 200), (102, 198), (101, 201), (103, 199)]
smoothed = sliding_window_average(raw_boxes)
```

## 面试常考

> **Q1: 数组和链表的区别？**

```
数组：连续内存，随机访问 O(1)，插入删除 O(n)
链表：分散内存，随机访问 O(n)，已知位置插入删除 O(1)
视觉中：存检测框用 vector（连续内存，cache 友好），管理 Tracklet 用链表
```

> **Q2: 什么场景用栈，什么场景用队列？**

```
栈（LIFO）：DFS、函数调用栈、撤销操作
队列（FIFO）：BFS、视频帧缓冲、消息队列
优先队列（堆）：NMS 候选框排序、Top-K 检测
```
