# 1.1.2 装饰器与上下文管理器

> **机器人视觉定位**：装饰器常用于**推理耗时统计**和**模型加载缓存**，上下文管理器用于**视频流/文件的安全释放**和**推理计时**。这两个模式写出的代码比手动 try/finally 优雅得多。

## 1. 装饰器 ⭐⭐

### 1.1 基础装饰器：推理耗时统计

```python
import time
import functools

def timeit(func):
    """装饰器：打印函数执行时间"""
    @functools.wraps(func)  # 保留原函数元信息
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        result = func(*args, **kwargs)
        elapsed = time.perf_counter() - start
        print(f"{func.__name__} 耗时: {elapsed:.3f}s")
        return result
    return wrapper

# 使用
@timeit
def inference(model, image):
    """模型推理（模拟）"""
    time.sleep(0.1)  # 模拟推理延迟
    return [{"label": "person", "score": 0.95}]

result = inference(None, None)  # 自动打印耗时
```

### 1.2 带参数的装饰器：灵活配置

```python
def repeat(n: int):
    """装饰器工厂：重复执行 n 次取平均"""
    def decorator(func):
        @functools.wraps(func)
        def wrapper(*args, **kwargs):
            times = []
            for _ in range(n):
                start = time.perf_counter()
                func(*args, **kwargs)
                times.append(time.perf_counter() - start)
            print(f"{func.__name__} {n}次平均耗时: {sum(times)/n:.3f}s")
        return wrapper
    return decorator

@repeat(n=5)
def preprocess(image):
    """图像预处理模拟"""
    _ = [i for i in range(1000000)]  # 模拟计算

preprocess(None)
```

### 1.3 视觉中的实用装饰器

```python
import functools
import time

class InferenceTimer:
    """类装饰器：累积推理时间统计"""
    def __init__(self, func):
        self.func = func
        self.total_time = 0.0
        self.call_count = 0
    
    def __call__(self, *args, **kwargs):
        start = time.perf_counter()
        result = self.func(*args, **kwargs)
        self.total_time += time.perf_counter() - start
        self.call_count += 1
        return result
    
    def report(self):
        print(f"调用 {self.call_count} 次，总耗时 {self.total_time:.3f}s，"
              f"平均 {self.total_time/self.call_count:.3f}s" if self.call_count > 0 else "未调用")

@InferenceTimer
def detect(frame):
    """目标检测推理"""
    time.sleep(0.05)  # 模拟推理
    return ["detection"]

# 使用
for _ in range(10):
    detect(None)
detect.report()
# 输出: 调用 10 次，总耗时 0.502s，平均 0.050s
```

## 2. 上下文管理器 ⭐⭐

### 2.1 基本用法：视频文件安全释放

```python
# ❌ 手动管理——容易忘记释放
cap = cv2.VideoCapture("video.mp4")
try:
    frames = []
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        frames.append(frame)
finally:
    cap.release()  # 容易忘记

# ✅ 上下文管理器——自动释放
import cv2

class VideoReader:
    """上下文管理器：自动释放视频资源"""
    def __init__(self, path: str):
        self.path = path
        self.cap = None
    
    def __enter__(self):
        self.cap = cv2.VideoCapture(self.path)
        if not self.cap.isOpened():
            raise RuntimeError(f"无法打开视频: {self.path}")
        return self.cap
    
    def __exit__(self, exc_type, exc_val, exc_tb):
        if self.cap is not None:
            self.cap.release()

# 使用——with 块结束后自动 release
with VideoReader("test.mp4") as cap:
    total_frames = int(cap.get(cv2.CAP_PROP_FRAME_COUNT))
    print(f"视频总帧数: {total_frames}")
```

### 2.2 推理计时上下文

```python
import time

class TimingContext:
    """上下文管理器：测量代码块执行时间"""
    def __enter__(self):
        self.start = time.perf_counter()
        return self
    
    def __exit__(self, *args):
        self.elapsed = time.perf_counter() - self.start
        print(f"耗时: {self.elapsed:.3f}s")

# 使用——对比不同实现的性能
with TimingContext() as t:
    # 方法 A：for 循环
    result_a = [i * 2 for i in range(1000000)]
print(f"列表推导式耗时: {t.elapsed:.3f}s")

with TimingContext():
    # 方法 B：NumPy 向量化
    import numpy as np
    result_b = np.arange(1000000) * 2
print(f"NumPy 向量化耗时: {t.elapsed:.3f}s")  # 通常快 10-50 倍
```

### 2.3 `contextlib` 简化写法

```python
from contextlib import contextmanager

@contextmanager
def video_capture(path: str):
    """用生成器实现上下文管理器（更简洁）"""
    cap = cv2.VideoCapture(path)
    try:
        yield cap
    finally:
        cap.release()

# 使用
with video_capture("test.mp4") as cap:
    ret, frame = cap.read()
    print(f"读取一帧，shape={frame.shape}")
```

## 3. 视觉中的典型应用

### 3.1 模型加载缓存装饰器

```python
import torch

def singleton_model(func):
    """装饰器：保证模型只加载一次（单例模式）"""
    instance = {}
    
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        key = str(id(func))
        if key not in instance:
            print("首次加载模型...")
            instance[key] = func(*args, **kwargs)
        return instance[key]
    return wrapper

@singleton_model
def load_yolo_model(weights_path: str):
    """加载 YOLO 模型"""
    model = torch.hub.load('ultralytics/yolov5', 'custom', path=weights_path)
    return model

# 多次调用只加载一次
model1 = load_yolo_model("yolov5s.pt")  # 实际加载
model2 = load_yolo_model("yolov5s.pt")  # 直接返回缓存
assert model1 is model2  # 同一个对象 ✅
```

## 必须掌握的要点

1. **`@functools.wraps`**：写装饰器必须加，保留原函数的 `__name__` 和 `__doc__`
2. **`with` 语句**：文件、视频流、网络资源等必须用 `with` 管理
3. **`__enter__`/`__exit__`**：能自定义上下文管理器（或 `@contextmanager` 简化版）
4. **装饰器 + 上下文管理器**：二者结合可以写出非常优雅的性能分析工具
