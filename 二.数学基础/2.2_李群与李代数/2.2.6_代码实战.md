## 2. 手写核心函数（面试要求）

### 2.1 指数映射（Rodriguez 公式）

```python
def hat(v):
    """向量 → 反对称矩阵"""
    return np.array([[0, -v[2], v[1]],
                     [v[2], 0, -v[0]],
                     [-v[1], v[0], 0]])

def vee(Phi):
    """反对称矩阵 → 向量"""
    return np.array([Phi[2, 1], Phi[0, 2], Phi[1, 0]])

def exp_so3(phi):
    """Rodriguez 公式：旋转向量 → 旋转矩阵"""
    theta = np.linalg.norm(phi)
    if theta < 1e-10:
        return np.eye(3)
    n = phi / theta
    n_hat = hat(n)
    return (np.eye(3) 
            + np.sin(theta) * n_hat 
            + (1 - np.cos(theta)) * (n_hat @ n_hat))

def log_so3(R):
    """旋转矩阵 → 旋转向量"""
    theta = np.arccos(np.clip((np.trace(R) - 1) / 2, -1.0, 1.0))
    if theta < 1e-10:
        return np.zeros(3)
    n_hat = (R - R.T) / (2 * np.sin(theta))
    n = vee(n_hat)
    return theta * n
```

### 2.2 se(3) 指数映射

```python
def V_matrix(phi):
    """se(3) 指数映射中的 V 矩阵"""
    theta = np.linalg.norm(phi)
    if theta < 1e-10:
        return np.eye(3)
    n = phi / theta
    n_hat = hat(n)
    return (np.eye(3) 
            + (1 - np.cos(theta)) / theta * n_hat 
            + (theta - np.sin(theta)) / theta * (n_hat @ n_hat))

def exp_se3(xi):
    """se(3) → SE(3) 指数映射"""
    rho, phi = xi[:3], xi[3:]
    R = exp_so3(phi)
    V = V_matrix(phi)
    t = V @ rho
    T = np.eye(4)
    T[:3, :3] = R
    T[:3, 3] = t
    return T
```

### 2.3 验证互逆性

```python
# 随机测试：指数映射 ↔ 对数映射
np.random.seed(42)
success = 0
for _ in range(1000):
    phi = np.random.randn(3) * 2.0
    R = exp_so3(phi)
    phi_back = log_so3(R)
    # 归一化处理（角度周期性）
    phi_norm = phi.copy()
    if np.linalg.norm(phi) > np.pi:
        phi_norm = phi / np.linalg.norm(phi) * (np.linalg.norm(phi) % np.pi)
    if np.allclose(phi_back, phi_norm, atol=1e-6):
        success += 1
print(f"互逆性通过率: {success}/1000")
```

## 4. BA 优化中的扰动模型实战

以下代码展示了 BA 优化中**位姿 Jacobian 的计算**，这是面试手撕的最高频场景：

```python
def ba_jacobian_full(T_wc, P_w, K):
    """
    计算重投影误差对 se(3) 位姿扰动的 Jacobian
    T_wc: (4,4) 世界→相机的变换
    P_w: (3,) 世界坐标系下的 3D 点
    K: (3,3) 内参矩阵
    
    return: J (2,6) Jacobian 矩阵
    """
    # 1. 3D 点转到相机坐标系
    R = T_wc[:3, :3]
    t = T_wc[:3, 3]
    P_c = R @ P_w + t          # (3,)
    
    X, Y, Z = P_c
    fx, fy = K[0, 0], K[1, 1]
    
    # 2. 重投影误差对 P_c 的导数 (2×3)
    # u = fx * X/Z + cx, v = fy * Y/Z + cy
    de_dPc = np.array([
        [fx/Z,     0, -fx*X/(Z*Z)],
        [0,     fy/Z, -fy*Y/(Z*Z)]
    ])
    
    # 3. P_c 对 se(3) 扰动的导数 (3×6)
    # ∂P_c / ∂Δξ = [E, -P_c^∧]
    dPc_dxi = np.hstack([np.eye(3), -hat(P_c)])
    
    # 4. 链式法则得到最终 Jacobian (2×6)
    J = de_dPc @ dPc_dxi
    return J


# 测试 BA Jacobian
K = np.array([[500, 0, 320],
              [0, 500, 240],
              [0,   0,   1]])
T_wc = np.eye(4)
T_wc[:3, :3] = exp_so3(np.array([0.1, -0.2, 0.3]))
T_wc[:3, 3] = np.array([0.5, -0.3, 2.0])

P_w = np.array([1.5, -0.5, 3.0])

J = ba_jacobian_full(T_wc, P_w, K)
print(f"BA Jacobian shape: {J.shape}")  # (2, 6)
print(f"BA Jacobian:\n{J}")
```

## 5. 完整示例：相机位姿优化

```python
def optimize_pose(T_init, pts_3d, pts_2d, K, n_iter=10):
    """
    用高斯-牛顿法优化相机位姿（简化版 BA）
    T_init: (4,4) 初始位姿估计
    pts_3d: (N, 3) 世界坐标系 3D 点
    pts_2d: (N, 2) 对应的观测像素
    K: (3,3) 内参
    """
    T = T_init.copy()
    
    for it in range(n_iter):
        J_total = []   # 所有点的 Jacobian 堆叠
        e_total = []   # 所有点的残差堆叠
        
        for i in range(len(pts_3d)):
            P_w = pts_3d[i]
            u_obs = pts_2d[i]
            
            # 投影
            P_c = T[:3, :3] @ P_w + T[:3, 3]
            u_proj = K @ (P_c / P_c[2])
            u_proj = u_proj[:2]
            
            # 残差
            e = u_obs - u_proj
            
            # Jacobian (2×6)
            J = ba_jacobian_full(T, P_w, K)
            
            J_total.append(J)
            e_total.append(e)
        
        # 堆叠所有点
        J = np.vstack(J_total)  # (2N, 6)
        e = np.hstack(e_total)  # (2N,)
        
        # 高斯-牛顿：Δξ = -(J^T J)⁻¹ J^T e
        H = J.T @ J
        b = J.T @ e
        delta_xi = np.linalg.solve(H, b)
        
        # 更新位姿（右乘扰动）
        T = T @ exp_se3(delta_xi)
        
        # 收敛判断
        if np.linalg.norm(delta_xi) < 1e-6:
            print(f"迭代 {it+1} 次后收敛")
            break
    
    return T
```
